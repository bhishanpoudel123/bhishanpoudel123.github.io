# Pandas dt vs isocalendar
# isocalendar always starts from Monday, not the actual start.
pd.Timestamp('2022-01-01').year                # 2022
pd.Timestamp('2022-01-01').isocalendar().year  # 2021 (iso calendar starts from Monday)
