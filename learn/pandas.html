3. Inspecting a DataFrame

df.head()          # first 5 rows
df.tail(3)         # last 3 rows
df.sample(5)       # random 5 rows
df.shape           # (rows, cols)
df.columns         # column names
df.dtypes          # column types
df.info()          # summary: types + non-null counts
df.describe()      # numeric summary stats
df.isna().sum()    # count missing per column
df["col"].value_counts()   # frequency of each value

4. Selecting rows and columns

# Column (Series)
df["name"]
df[["name", "score"]]      # multiple columns → DataFrame

# By position (iloc)
df.iloc[0]                 # first row
df.iloc[0:5, 0:3]          # first 5 rows, first 3 cols
df.iloc[:, -1]             # last column

# By label (loc)
df.loc[10]                 # row with index 10
df.loc[10, "name"]         # single cell
df.loc[10:20, ["name", "score"]]

# Reset / set index
df.reset_index(drop=True, inplace=True)
df.set_index("id", inplace=True)

5. Filtering

df[df["score"] > 80]
df[(df["score"] > 80) & (df["gender"] == "F")]
df[df["city"].isin(["NY", "LA", "SF"])]
df[df["name"].str.contains("smith", case=False, na=False)]
df[df["score"].between(70, 90)]

# Query syntax (cleaner for complex filters)
df.query("score > 80 and gender == 'F'")

6. Missing values

df.isna().sum()                # count per column
df.dropna()                    # drop rows with any NaN
df.dropna(subset=["score"])    # drop rows where score is NaN
df.fillna(0)                   # replace NaN with 0
df["score"].fillna(df["score"].mean(), inplace=True)
df["city"].fillna("Unknown", inplace=True)

# Forward / backward fill (good for time series)
df["price"].ffill(inplace=True)
df["price"].bfill(inplace=True)

7. GroupBy & aggregation

df.groupby("city")["score"].mean()
df.groupby("city")["score"].agg(["mean", "median", "count"])
df.groupby(["city", "gender"])["score"].sum()

# Multiple aggregations with named output
df.groupby("city").agg(
    avg_score=("score", "mean"),
    max_score=("score", "max"),
    n=("score", "size"),
).reset_index()

# Pivot table
pd.pivot_table(df, index="city", columns="gender",
               values="score", aggfunc="mean")

8. Merge & join

# Inner join on a common column
pd.merge(orders, customers, on="customer_id", how="inner")

# Different column names
pd.merge(orders, customers,
         left_on="cust_id", right_on="id", how="left")

# Concatenate rows
pd.concat([df1, df2], ignore_index=True)

# Concatenate columns
pd.concat([df1, df2], axis=1)

9. Dates & time

df["date"] = pd.to_datetime(df["date"])
df["year"]  = df["date"].dt.year
df["month"] = df["date"].dt.month
df["day"]   = df["date"].dt.day
df["dow"]   = df["date"].dt.day_name()

# Filter by date range
df[df["date"] >= "2024-01-01"]

# Resample time series (e.g., monthly mean)
df.set_index("date").resample("M")["value"].mean()

10. Saving data

df.to_csv("out.csv", index=False)
df.to_excel("out.xlsx", index=False)
df.to_json("out.json", orient="records")
df.to_parquet("out.parquet", index=False)
